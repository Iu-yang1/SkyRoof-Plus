cmake_minimum_required(VERSION 3.20)
project(skyroof_media_native VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPUS REQUIRED IMPORTED_TARGET opus)
add_library(skyroof_media_native SHARED src/skyroof_opus.cpp)
target_compile_definitions(skyroof_media_native PRIVATE SKYROOF_MEDIA_BUILD)
target_include_directories(skyroof_media_native PUBLIC include)
target_link_libraries(skyroof_media_native PRIVATE PkgConfig::OPUS)

include(CTest)
if(BUILD_TESTING)
    add_executable(skyroof_opus_test tests/opus_roundtrip.cpp)
    target_link_libraries(skyroof_opus_test PRIVATE skyroof_media_native)
    add_test(NAME opus_roundtrip COMMAND skyroof_opus_test)
endif()
