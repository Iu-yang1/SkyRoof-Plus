cmake_minimum_required(VERSION 3.20)
project(skyroof_rtc LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED LIBDATACHANNEL_SOURCE)
  message(FATAL_ERROR "Pass the pinned libdatachannel tree")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_MEDIA OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${LIBDATACHANNEL_SOURCE}" "${CMAKE_BINARY_DIR}/rtc-dependency" EXCLUDE_FROM_ALL)
find_path(OPUS_INCLUDE_DIR opus/opus.h REQUIRED)
find_library(OPUS_LIBRARY NAMES opus REQUIRED)
add_library(skyroof_rtc SHARED rtc_rx_peer.cpp rtc_voice_track.cpp)
target_compile_definitions(skyroof_rtc PRIVATE SKYROOF_RTC_BUILD)
target_link_libraries(skyroof_rtc PRIVATE datachannel-static ${OPUS_LIBRARY})
target_include_directories(skyroof_rtc PRIVATE "${LIBDATACHANNEL_SOURCE}/include" ${OPUS_INCLUDE_DIR})
include(CTest)
if(BUILD_TESTING)
  add_executable(skyroof_rtc_rx_test rtc_rx_test.cpp)
  target_link_libraries(skyroof_rtc_rx_test PRIVATE skyroof_rtc datachannel-static)
  target_include_directories(skyroof_rtc_rx_test PRIVATE "${LIBDATACHANNEL_SOURCE}/include")
  add_test(NAME rtc_rx_bridge COMMAND skyroof_rtc_rx_test)
  set_tests_properties(rtc_rx_bridge PROPERTIES TIMEOUT 40)
endif()
