cmake_minimum_required(VERSION 3.20)
project(skyroof_webrtc_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED LIBDATACHANNEL_SOURCE)
  message(FATAL_ERROR "Pass -DLIBDATACHANNEL_SOURCE to the pinned official tree")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_MEDIA OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBDATACHANNEL_SOURCE}" "${CMAKE_BINARY_DIR}/rtc-dependency" EXCLUDE_FROM_ALL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPUS REQUIRED IMPORTED_TARGET opus)
add_executable(skyroof_webrtc_loopback datachannel_loopback.cpp)
target_link_libraries(skyroof_webrtc_loopback PRIVATE datachannel-static PkgConfig::OPUS)
target_include_directories(skyroof_webrtc_loopback PRIVATE "${LIBDATACHANNEL_SOURCE}/include")
enable_testing()
add_test(NAME webrtc_datachannel_loopback COMMAND skyroof_webrtc_loopback)
set_tests_properties(webrtc_datachannel_loopback PROPERTIES TIMEOUT 45)
